def read_text_file(path)
    encodings = [utf-8-sig, utf-8, cp949]
    for enc in encodings
        try
            with open(path, r, encoding=enc, errors=strict) as f return f.read()
        except UnicodeDecodeError continue
    raise Exception(f無法識別的檔案編碼：{path})

def write_text_file(path, content, mode=w)
    with open(path, mode, encoding=utf-8, newline=) as f f.write(content)

def load_json(path)
    if not os.path.exists(path) return {}
    try
        with open(path, r, encoding=utf-8) as f
            c = f.read().strip()
            return json.loads(c) if c else {}
    except return {}

def save_json(path, data)
    tmp = path + .tmp
    try
        with open(tmp, w, encoding=utf-8) as f json.dump(data, f, ensure_ascii=False, indent=4)
        os.replace(tmp, path)
    except pass

def split_text(text)
    chunks = []
    text = text.replace(rn, n).strip()
    while len(text)  CHUNK_SIZE
        cut = CHUNK_SIZE
        separators = [nn, n, 。, ！, ？, ., !, ]
        for sep in separators
            pos = text.rfind(sep, 0, CHUNK_SIZE)
            if pos  CHUNK_SIZE  0.5
                cut = pos + len(sep)
                break
        chunk = text[cut].strip()
        if chunk chunks.append(chunk)
        text = text[cut].strip()
    if text chunks.append(text)
    return chunks

def check_translation_basic(text)
    if not text return False, 空白內容
    if len(text)  5 return False, 內容過短
    bad_words = [以下是翻譯, 翻譯如下, Here is the translation]
    for word in bad_words
        if word in text return False, 包含 AI 說明文字
    return True, 正常
